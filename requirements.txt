# BrandStudio Flask backend — pinned to Python 3.12.
#
# Pillow 10.1.0 and cryptography 42.0.8 only publish prebuilt wheels through
# Python 3.12. On 3.13 / 3.14, `pip install` here falls back to building from
# source ("Getting requirements to build wheel ... error") which needs MSVC +
# libjpeg + Rust + OpenSSL locally. To upgrade Python:
#   1. Bump Pillow to 11.x (has 3.13 / 3.14 wheels)
#   2. Bump cryptography to 44.x (has 3.13 / 3.14 wheels)
#   3. Re-test the imports in app.py and lib/cloudflare_dns.py.
# Until then: use `py -3.12` on Windows or `python3.12` on Linux/macOS.
Flask==3.0.0
Werkzeug==3.0.1
Pillow==10.1.0
PyJWT==2.8.0
requests==2.31.0
PyMySQL==1.1.0
python-dotenv==1.0.1
cryptography==42.0.8
# Playwright — real-browser fetcher used by /api/extract-website and
# /api/ai/brand for sites that block plain HTTP scrapers (AutoTrader behind
# Cloudflare, JS-rendered SPAs). After `pip install -r requirements.txt`:
#     python -m playwright install chromium --no-shell
#     python -m playwright install-deps chromium    # Linux only
# IMPORTANT: --no-shell installs the FULL Chromium browser. The default
# `install chromium` in 1.50+ downloads only chromium-headless-shell, which
# DOES NOT support HTTP proxy authentication (you'll get
# ERR_PROXY_AUTH_UNSUPPORTED with iProyal / Bright Data / any authed proxy).
# Downloads ~150 MB on first install.
playwright==1.50.0
